O QUE É O SUCESSO DA LOGIN.SERVICE, PARA QUE SERVE


app.post("/login", async (req, res) => {
  try {
    const { email, senha } = req.body;
    const [usuarios] = await env.db.query(
      "SELECT * FROM USUARIOS WHERE EMAIL = ?",
      [email],
    );
    if (usuarios.length === 0) {
      return res.status(404).json({
        erro: "Usuário não encontrado",
      });
    }
    const usuario = usuarios[0]; //começa a lista no primeiro termo

    if (usuario.SENHA !== senha) {
      return res.status(401).json({
        erro: "senha incorreta",
      });
    }
    //aqui é quando der certo
    res.status(200).json({
      mensagem: "login realizado com sucesso",
      //depois apagar isso
      usuario: {
        id: usuario.ID,
        nome: usuario.NOME,
        email: usuario.EMAIL,
      },
    });
  } catch (erro) {
    console.error(erro);
    res.status(500).json({
      erro: "erro ao realizar login ",
    });
  }
});
